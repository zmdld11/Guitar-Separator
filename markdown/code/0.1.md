import os
import torch
import torchaudio
import numpy as np
from torch.utils.data import Dataset, DataLoader
import json
import random
from pathlib import Path
import yaml
from tqdm import tqdm
import warnings
import re
import hashlib
import pickle

class GuitarSeparationDataset(Dataset):
    def __init__(self, config, mode="train", use_cache=True):
        self.config = config
        self.mode = mode
        self.sample_rate = config.sample_rate
        self.duration = config.duration
        
        # 确保样本数是hop_length的倍数，避免STFT尺寸问题
        target_samples = int(self.duration * self.sample_rate)
        # 调整到最近的hop_length倍数
        self.samples_per_clip = (target_samples // config.hop_length) * config.hop_length
        
        print(f"音频片段长度: {self.samples_per_clip} 样本 ({self.samples_per_clip/self.sample_rate:.2f} 秒)")
        
        # 修复缓存路径
        project_root = Path(__file__).parent.parent
        self.cache_dir = project_root / "data" / "cache"
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.cache_file = self.cache_dir / f"dataset_cache_{mode}.pkl"
        
        print("正在加载数据集...")
        
        if use_cache and self.cache_file.exists():
            print("从缓存加载数据集...")
            self.samples = self._load_from_cache()
        else:
            self.samples = self._load_samples()
            if use_cache:
                self._save_to_cache()
        
        if len(self.samples) == 0:
            warnings.warn("没有找到任何样本，请检查数据集路径")
        else:
            print(f"成功加载 {len(self.samples)} 个样本")
    
    def _load_from_cache(self):
        """从缓存加载数据"""
        try:
            with open(self.cache_file, 'rb') as f:
                cache_data = pickle.load(f)
            
            if self._validate_cache(cache_data):
                print("缓存验证成功")
                return cache_data['samples']
            else:
                print("缓存已过期，重新扫描数据集")
                return self._load_samples()
        except Exception as e:
            print(f"加载缓存失败: {e}，重新扫描数据集")
            return self._load_samples()
    
    def _save_to_cache(self):
        """保存数据到缓存"""
        try:
            cache_data = {
                'samples': self.samples,
                'config_hash': self._get_config_hash(),
                'dataset_modified': self._get_dataset_modified_time()
            }
            with open(self.cache_file, 'wb') as f:
                pickle.dump(cache_data, f)
            print(f"数据集缓存已保存: {self.cache_file}")
        except Exception as e:
            print(f"保存缓存失败: {e}")
    
    def _validate_cache(self, cache_data):
        """验证缓存是否有效"""
        if cache_data.get('config_hash') != self._get_config_hash():
            return False
        
        cached_modified = cache_data.get('dataset_modified', {})
        current_modified = self._get_dataset_modified_time()
        
        for dataset, modified_time in current_modified.items():
            if cached_modified.get(dataset) != modified_time:
                return False
        
        return True
    
    def _get_config_hash(self):
        """获取配置的哈希值"""
        config_str = f"{self.sample_rate}_{self.duration}_{self.config.moisesdb_dir}_{self.config.medleydb_dir}"
        return hashlib.md5(config_str.encode()).hexdigest()
    
    def _get_dataset_modified_time(self):
        """获取数据集的最后修改时间"""
        modified_times = {}
        
        moises_path = Path(self.config.moisesdb_dir)
        if moises_path.exists():
            try:
                all_files = [f for f in moises_path.rglob("*") if f.is_file()]
                if all_files:
                    latest_moises = max(f.stat().st_mtime for f in all_files)
                    modified_times['moisesdb'] = latest_moises
                else:
                    modified_times['moisesdb'] = 0
            except Exception as e:
                print(f"获取MoisesDB修改时间出错: {e}")
                modified_times['moisesdb'] = 0
        
        medley_path = Path(self.config.medleydb_dir)
        if medley_path.exists():
            try:
                all_files = [f for f in medley_path.rglob("*") if f.is_file()]
                if all_files:
                    latest_medley = max(f.stat().st_mtime for f in all_files)
                    modified_times['medleydb'] = latest_medley
                else:
                    modified_times['medleydb'] = 0
            except Exception as e:
                print(f"获取MedleyDB修改时间出错: {e}")
                modified_times['medleydb'] = 0
        
        return modified_times
    
    def _load_moisesdb_samples(self):
        """加载MoisesDB数据集样本"""
        samples = []
        moises_path = Path(self.config.moisesdb_dir)
        
        if not moises_path.exists():
            print(f"警告: MoisesDB路径不存在: {moises_path}")
            return samples
        
        print("正在扫描MoisesDB数据集...")
        item_dirs = [d for d in moises_path.iterdir() if d.is_dir()]
        
        for item_dir in tqdm(item_dirs, desc="MoisesDB"):
            try:
                data_json_path = item_dir / "data.json"
                if not data_json_path.exists():
                    continue
                
                with open(data_json_path, 'r', encoding='utf-8') as f:
                    data_info = json.load(f)
                
                guitar_dir = item_dir / "guitar"
                if guitar_dir.exists() and guitar_dir.is_dir():
                    mix_audio_path = self._create_mix_from_stems(item_dir, data_info)
                    if mix_audio_path is not None:
                        guitar_files = list(guitar_dir.glob("*.wav"))
                        for guitar_file in guitar_files:
                            if guitar_file.exists():
                                samples.append({
                                    'mix': str(mix_audio_path),
                                    'guitar': str(guitar_file),
                                    'is_temp_mix': True
                                })
            except Exception as e:
                print(f"加载MoisesDB样本时出错 {item_dir}: {e}")
                continue
        
        print(f"从MoisesDB加载了 {len(samples)} 个样本")
        return samples
    
    def _create_mix_from_stems(self, item_dir, data_info):
        """从各个音轨创建混合音轨"""
        temp_mix_path = item_dir / "temp_mix.wav"
        
        if temp_mix_path.exists() and temp_mix_path.stat().st_size > 1000:
            return temp_mix_path
            
        try:
            all_stems = []
            stem_dirs = ['guitar', 'bass', 'drums', 'vocals', 'piano', 'other']
            
            for stem_dir in stem_dirs:
                stem_path = item_dir / stem_dir
                if stem_path.exists() and stem_path.is_dir():
                    for stem_file in stem_path.glob("*.wav"):
                        try:
                            audio, sr = torchaudio.load(stem_file)
                            if sr != self.sample_rate:
                                resampler = torchaudio.transforms.Resample(sr, self.sample_rate)
                                audio = resampler(audio)
                            all_stems.append(audio)
                        except Exception as e:
                            print(f"加载音轨 {stem_file} 时出错: {e}")
                            continue
            
            if len(all_stems) == 0:
                print(f"警告: {item_dir} 中没有找到有效的音轨文件")
                return None
            
            max_length = max(stem.shape[1] for stem in all_stems)
            
            mix_audio = torch.zeros((2, max_length))
            for stem in all_stems:
                if stem.shape[0] == 1:
                    stem = stem.repeat(2, 1)
                elif stem.shape[0] > 2:
                    stem = stem[:2, :]
                
                if stem.shape[1] < max_length:
                    padded_stem = torch.zeros((2, max_length))
                    padded_stem[:, :stem.shape[1]] = stem
                    stem = padded_stem
                
                mix_audio += stem
            
            max_val = mix_audio.abs().max()
            if max_val > 0:
                mix_audio = mix_audio / (max_val * 1.1)
            
            torchaudio.save(str(temp_mix_path), mix_audio, self.sample_rate)
            print(f"创建混合文件: {temp_mix_path}")
            
            return temp_mix_path
            
        except Exception as e:
            print(f"创建混合音轨时出错 {item_dir}: {e}")
            return None
    
    def _load_medleydb_samples(self):
        """加载MedleyDB数据集样本"""
        samples = []
        medley_path = Path(self.config.medleydb_dir)
        
        if not medley_path.exists():
            print(f"警告: MedleyDB路径不存在: {medley_path}")
            return samples
        
        print("正在扫描MedleyDB数据集...")
        track_dirs = [d for d in medley_path.iterdir() if d.is_dir()]
        
        for track_dir in tqdm(track_dirs, desc="MedleyDB"):
            try:
                mix_files = list(track_dir.glob("*_MIX.wav"))
                if not mix_files:
                    continue
                    
                mix_path = mix_files[0]
                stems_dir = track_dir / f"{track_dir.name}_STEMS"
                
                if not stems_dir.exists():
                    continue
                    
                for stem_file in stems_dir.glob("*.wav"):
                    if self._is_guitar_stem(stem_file.name):
                        samples.append({
                            'mix': str(mix_path),
                            'guitar': str(stem_file),
                            'is_temp_mix': False
                        })
            except Exception as e:
                print(f"处理MedleyDB目录 {track_dir} 时出错: {e}")
                continue
        
        print(f"从MedleyDB加载了 {len(samples)} 个样本")
        return samples
    
    def _is_guitar_stem(self, filename):
        """识别吉他音轨"""
        guitar_keywords = [
            'guitar', 'gtr', 'git', 'guitare', 'guitarra', 
            'acoustic', 'electric', 'nylon', 'steel'
        ]
        anti_keywords = ['bass', 'kick', 'snare', 'drum', 'vocal', 'voice']
        
        filename_lower = filename.lower()
        
        for anti in anti_keywords:
            if anti in filename_lower:
                return False
        
        for keyword in guitar_keywords:
            if keyword in filename_lower:
                return True
        
        stem_match = re.search(r'STEM_(\d+)', filename)
        if stem_match:
            stem_num = int(stem_match.group(1))
            if stem_num <= 4:
                return True
        
        return False
    
    def _load_samples(self):
        """加载所有样本"""
        samples = []
        
        moises_samples = self._load_moisesdb_samples()
        samples.extend(moises_samples)
        
        medley_samples = self._load_medleydb_samples()
        samples.extend(medley_samples)
        
        if len(samples) == 0:
            print("创建测试样本...")
            samples = self._create_test_samples()
        
        return samples
    
    def _create_test_samples(self):
        """创建测试样本（用于调试）"""
        test_samples = []
        project_root = Path(__file__).parent.parent
        test_dir = project_root / "test_samples"
        test_dir.mkdir(exist_ok=True)
        
        print("正在生成测试音频...")

            mix_audio = torch.zeros(2, len(t))
            for freq in mix_freqs:
                wave = torch.sin(2 * np.pi * freq * t)
                mix_audio += wave.unsqueeze(0).repeat(2, 1) * 0.2
            
            guitar_freqs = [220, 440, 660]
            guitar_audio = torch.zeros(2, len(t))
            for freq in guitar_freqs:
                wave = torch.sin(2 * np.pi * freq * t)
                guitar_audio += wave.unsqueeze(0).repeat(2, 1) * 0.4
            
            envelope = torch.ones_like(t)
            attack = int(0.1 * len(t))
            release = int(0.2 * len(t))
            envelope[:attack] = torch.linspace(0, 1, attack)
            envelope[-release:] = torch.linspace(1, 0, release)
            
            mix_audio *= envelope
            guitar_audio *= envelope
            
            mix_path = test_dir / f"test_mix_{i:02d}.wav"
            guitar_path = test_dir / f"test_guitar_{i:02d}.wav"
            
            torchaudio.save(str(mix_path), mix_audio, self.sample_rate)
            torchaudio.save(str(guitar_path), guitar_audio, self.sample_rate)
            
            test_samples.append({
                'mix': str(mix_path),
                'guitar': str(guitar_path),
                'is_temp_mix': False
            })
        
        print(f"创建了 {len(test_samples)} 个测试样本")
        return test_samples
    
    def _load_audio(self, path):
        try:
            audio, sr = torchaudio.load(path)
            if sr != self.sample_rate:
                resampler = torchaudio.transforms.Resample(sr, self.sample_rate)
                audio = resampler(audio)
            
            if audio.shape[0] == 1:
                audio = audio.repeat(2, 1)
            elif audio.shape[0] > 2:
                audio = audio[:2, :]
                
            return audio
        except Exception as e:
            print(f"加载音频出错 {path}: {e}")
            return torch.zeros((2, self.samples_per_clip))
    
    def __len__(self):
        return len(self.samples)
    
    def __getitem__(self, idx):
        sample = self.samples[idx]
        
        mix_audio = self._load_audio(sample['mix'])
        guitar_audio = self._load_audio(sample['guitar'])
        
        min_length = min(mix_audio.shape[1], guitar_audio.shape[1])
        mix_audio = mix_audio[:, :min_length]
        guitar_audio = guitar_audio[:, :min_length]
        
        # 确保长度是hop_length的倍数
        if mix_audio.shape[1] > self.samples_per_clip:
            # 裁剪到目标长度（确保是hop_length的倍数）
            cropped_length = (self.samples_per_clip // self.config.hop_length) * self.config.hop_length
            start = random.randint(0, mix_audio.shape[1] - cropped_length)
            mix_audio = mix_audio[:, start:start + cropped_length]
            guitar_audio = guitar_audio[:, start:start + cropped_length]
        elif mix_audio.shape[1] < self.samples_per_clip:
            # 填充到目标长度（确保是hop_length的倍数）
            padded_length = (self.samples_per_clip // self.config.hop_length) * self.config.hop_length
            padded_mix = torch.zeros((2, padded_length))
            padded_guitar = torch.zeros((2, padded_length))
            padded_mix[:, :mix_audio.shape[1]] = mix_audio
            padded_guitar[:, :guitar_audio.shape[1]] = guitar_audio
            mix_audio = padded_mix
            guitar_audio = padded_guitar
        
        if self.mode == "train":
            gain = random.uniform(0.7, 1.3)
            mix_audio = mix_audio * gain
            guitar_audio = guitar_audio * gain
            
            if random.random() > 0.5:
                mix_audio = -mix_audio
                guitar_audio = -guitar_audio
            
            if random.random() > 0.8:
                noise_level = random.uniform(0.001, 0.01)
                mix_audio += torch.randn_like(mix_audio) * noise_level
        
        return {
            'mix': mix_audio,
            'guitar': guitar_audio
        }

def get_dataloaders(config):
    print("初始化数据加载器...")
    dataset = GuitarSeparationDataset(config, mode="train", use_cache=True)
    
    if len(dataset) == 0:
        raise ValueError("没有找到任何训练样本，请检查数据集路径")
    
    train_size = int(0.8 * len(dataset))
    val_size = len(dataset) - train_size
    
    if train_size == 0 or val_size == 0:
        train_dataset = dataset
        val_dataset = dataset
        print("使用全部数据进行训练和验证")
    else:
        train_dataset, val_dataset = torch.utils.data.random_split(
            dataset, [train_size, val_size]
        )
        print(f"训练集: {len(train_dataset)} 样本, 验证集: {len(val_dataset)} 样本")
    
    train_loader = DataLoader(
        train_dataset,
        batch_size=config.batch_size,
        shuffle=True,
        num_workers=min(config.num_workers, 4),
        pin_memory=True
    )
    
    val_loader = DataLoader(
        val_dataset,
        batch_size=config.batch_size,
        shuffle=False,
        num_workers=min(config.num_workers, 4),
        pin_memory=True
    )
    
    return train_loader, val_loader

import torch
import torch.nn as nn
import torch.nn.functional as F
import numpy as np

class FrequencyUNet(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.config = config
        
        # STFT参数
        self.n_fft = config.n_fft
        self.hop_length = config.hop_length
        self.register_buffer('window', torch.hann_window(self.n_fft))
        
        channels = config.unet_channels
        
        # 编码器
        self.encoders = nn.ModuleList()
        for i in range(len(channels)):
            in_ch = 4 if i == 0 else channels[i-1]  # 4个通道：实部1，实部2，虚部1，虚部2
            out_ch = channels[i]
            
            encoder = nn.Sequential(
                nn.Conv2d(in_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU(),
                nn.Conv2d(out_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU()
            )
            self.encoders.append(encoder)
        
        # 下采样层
        self.downsample_layers = nn.ModuleList([
            nn.MaxPool2d(2) for _ in range(len(channels) - 1)
        ])
        
        # 解码器
        self.decoders = nn.ModuleList()
        for i in range(len(channels)-1, 0, -1):
            in_ch = channels[i] + channels[i-1]  # 跳跃连接
            out_ch = channels[i-1]
            
            decoder = nn.Sequential(
                nn.Conv2d(in_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU(),
                nn.Conv2d(out_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU()
            )
            self.decoders.append(decoder)
        
        # 上采样层
        self.upsample_layers = nn.ModuleList([
            nn.Upsample(scale_factor=2, mode='bilinear', align_corners=False) 
            for _ in range(len(channels) - 1)
        ])
        
        # 输出层 - 预测吉他掩码
        self.output = nn.Conv2d(channels[0], 2, 1)  # 2个通道：实部掩码，虚部掩码
    
    def forward(self, x):
        batch_size, audio_channels, samples = x.shape
        
        # 计算STFT
        spec_list = []
        for i in range(audio_channels):
            channel_spec = torch.stft(x[:, i], n_fft=self.n_fft, hop_length=self.hop_length,
                                    window=self.window, return_complex=True)
            spec_list.append(channel_spec)
        
        # 合并通道 [batch, 2, freq, time]
        spec_real = torch.stack([s.real for s in spec_list], dim=1)
        spec_imag = torch.stack([s.imag for s in spec_list], dim=1)
        
        # 输入特征 [batch, 4, freq, time]
        freq_input = torch.cat([spec_real, spec_imag], dim=1)
        
        # 编码路径
        skips = []
        current = freq_input
        
        for i, encoder in enumerate(self.encoders):
            current = encoder(current)
            skips.append(current)
            
            # 下采样（除了最后一层）
            if i < len(self.downsample_layers):
                current = self.downsample_layers[i](current)
        
        # 解码路径
        for i, (upsample, decoder) in enumerate(zip(self.upsample_layers, self.decoders)):
            # 上采样
            current = upsample(current)
            
            # 获取跳跃连接
            skip = skips[-(i+2)]
            
            # 调整尺寸以匹配
            if current.shape[2:] != skip.shape[2:]:
                current = F.interpolate(current, size=skip.shape[2:], mode='bilinear', align_corners=False)
            
            # 连接跳跃连接
            current = torch.cat([current, skip], dim=1)
            
            # 解码
            current = decoder(current)
        
        # 预测掩码
        mask = self.output(current)
        mask_real = mask[:, 0:1]  # [batch, 1, freq, time]
        mask_imag = mask[:, 1:2]  # [batch, 1, freq, time]
        
        # 应用掩码到原始频谱
        # 注意：这里我们只对第一个音频通道应用掩码，因为我们想要分离吉他
        output_spec_real = spec_real[:, 0:1] * mask_real - spec_imag[:, 0:1] * mask_imag
        output_spec_imag = spec_real[:, 0:1] * mask_imag + spec_imag[:, 0:1] * mask_real
        
        # 合并为复数频谱 [batch, 1, freq, time]
        output_spec = torch.complex(output_spec_real, output_spec_imag)
        
        # ISTFT恢复时域信号 - 注意维度处理
        output_wav_list = []
        for i in range(output_spec.shape[0]):
            # 对每个批次单独处理
            single_spec = output_spec[i, 0]  # [freq, time]
            single_wav = torch.istft(single_spec, n_fft=self.n_fft, hop_length=self.hop_length,
                                   length=samples, window=self.window)
            output_wav_list.append(single_wav)
        
        # 堆叠结果 [batch, samples]
        output_wav = torch.stack(output_wav_list, dim=0)
        
        return output_wav.unsqueeze(1)  # [batch, 1, samples]

class GuitarSeparator(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.freq_unet = FrequencyUNet(config)
        
    def forward(self, mix):
        return self.freq_unet(mix)

import torch
import torch.nn as nn
from torch.optim import Adam
from torch.optim.lr_scheduler import ReduceLROnPlateau
import torch.nn.functional as F
import time
import os
from pathlib import Path
from tqdm import tqdm

from config import Config
from dataset import get_dataloaders
from model import GuitarSeparator

class Trainer:
    def __init__(self, config):
        self.config = config
        self.device = config.device
        
        # 创建目录
        project_root = Path(__file__).parent.parent
        (project_root / "model").mkdir(exist_ok=True)
        (project_root / "output").mkdir(exist_ok=True)
        
        print("正在初始化模型...")
        # 初始化模型
        self.model = GuitarSeparator(config).to(self.device)
        
        # 打印模型参数数量
        total_params = sum(p.numel() for p in self.model.parameters())
        print(f"模型参数总数: {total_params:,}")
        
        # 优化器
        self.optimizer = Adam(self.model.parameters(), lr=config.learning_rate)
        self.scheduler = ReduceLROnPlateau(self.optimizer, mode='min', factor=0.5, patience=5, verbose=True)
        
        # 损失函数
        self.l1_loss = nn.L1Loss()
        self.mse_loss = nn.MSELoss()
        
        # 数据加载器
        self.train_loader, self.val_loader = get_dataloaders(config)
        
        # 训练状态
        self.best_val_loss = float('inf')
        self.train_losses = []
        self.val_losses = []
        
    def compute_loss(self, output, target):
        # 确保输出和目标形状一致
        # 如果目标是立体声，取平均值转为单声道
        if target.shape[1] == 2 and output.shape[1] == 1:
            target_mono = target.mean(dim=1, keepdim=True)  # 立体声转单声道
        else:
            target_mono = target
        
        # 基础时域损失
        l1_loss = self.l1_loss(output, target_mono)
        mse_loss = self.mse_loss(output, target_mono)
        
        # 频谱一致性损失 - 修复维度问题
        try:
            # 确保输入是2D [batch, samples]
            output_2d = output.squeeze(1)  # 移除通道维度 [batch, samples]
            target_2d = target_mono.squeeze(1)  # 移除通道维度 [batch, samples]
            
            output_spec = torch.stft(output_2d, n_fft=self.config.n_fft, 
                                hop_length=self.config.hop_length, 
                                return_complex=True, 
                                window=torch.hann_window(self.config.n_fft).to(output.device))
            target_spec = torch.stft(target_2d, n_fft=self.config.n_fft, 
                                hop_length=self.config.hop_length, 
                                return_complex=True, 
                                window=torch.hann_window(self.config.n_fft).to(target.device))
            
            magnitude_loss = F.l1_loss(output_spec.abs(), target_spec.abs())
            spec_loss = magnitude_loss
        except Exception as e:
            print(f"频谱损失计算失败: {e}")
            spec_loss = torch.tensor(0.0, device=output.device)
        
        # 总损失
        total_loss = l1_loss + 0.5 * mse_loss + 0.2 * spec_loss
        return total_loss
    
    def train_epoch(self, epoch):
        self.model.train()
        total_loss = 0
        num_batches = len(self.train_loader)
        
        progress_bar = tqdm(self.train_loader, desc=f'训练 Epoch {epoch}')
        
        for batch_idx, batch in enumerate(progress_bar):
            mix = batch['mix'].to(self.device)
            guitar = batch['guitar'].to(self.device)
            
            self.optimizer.zero_grad()
            
            try:
                output = self.model(mix)
                loss = self.compute_loss(output, guitar)
                
                loss.backward()
                torch.nn.utils.clip_grad_norm_(self.model.parameters(), max_norm=1.0)
                self.optimizer.step()
                
                total_loss += loss.item()
                
                # 更新进度条
                progress_bar.set_postfix({
                    'Loss': f'{loss.item():.6f}',
                    'Avg Loss': f'{total_loss / (batch_idx + 1):.6f}'
                })
            except Exception as e:
                print(f"训练批次 {batch_idx} 时出错: {e}")
                continue
        
        return total_loss / num_batches if num_batches > 0 else float('inf')
    
    def validate(self, epoch):
        self.model.eval()
        total_loss = 0
        num_batches = len(self.val_loader)
        
        progress_bar = tqdm(self.val_loader, desc=f'验证 Epoch {epoch}')
        
        with torch.no_grad():
            for batch_idx, batch in enumerate(progress_bar):
                mix = batch['mix'].to(self.device)
                guitar = batch['guitar'].to(self.device)
                
                try:
                    output = self.model(mix)
                    loss = self.compute_loss(output, guitar)
                    
                    total_loss += loss.item()
                    
                    # 更新进度条
                    progress_bar.set_postfix({
                        'Loss': f'{loss.item():.6f}',
                        'Avg Loss': f'{total_loss / (batch_idx + 1):.6f}'
                    })
                except Exception as e:
                    print(f"验证批次 {batch_idx} 时出错: {e}")
                    continue
        
        return total_loss / num_batches if num_batches > 0 else float('inf')
    
    def save_model(self, path):
        project_root = Path(__file__).parent.parent
        full_path = project_root / path
        full_path.parent.mkdir(parents=True, exist_ok=True)
        
        torch.save({
            'model_state_dict': self.model.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'best_val_loss': self.best_val_loss,
            'train_losses': self.train_losses,
            'val_losses': self.val_losses,
            'config': self.config.__dict__
        }, str(full_path))
    
    def load_model(self, path):
        project_root = Path(__file__).parent.parent
        full_path = project_root / path
        checkpoint = torch.load(full_path, map_location=self.device)
        self.model.load_state_dict(checkpoint['model_state_dict'])
        self.optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
        self.best_val_loss = checkpoint['best_val_loss']
        self.train_losses = checkpoint['train_losses']
        self.val_losses = checkpoint['val_losses']
    
    def log(self, message):
        print(message)
        project_root = Path(__file__).parent.parent
        log_path = project_root / self.config.log_path
        log_path.parent.mkdir(parents=True, exist_ok=True)
        with open(log_path, 'a', encoding='utf-8') as f:
            f.write(message + '\n')
    
    def train(self):
        self.log("开始训练吉他音轨分离模型...")
        self.log(f"设备: {self.device}")
        self.log(f"训练样本数: {len(self.train_loader.dataset)}")
        self.log(f"验证样本数: {len(self.val_loader.dataset)}")
        self.log(f"批次大小: {self.config.batch_size}")
        self.log(f"学习率: {self.config.learning_rate}")
        
        for epoch in range(1, self.config.epochs + 1):
            start_time = time.time()
            
            # 训练
            train_loss = self.train_epoch(epoch)
            
            # 验证
            val_loss = self.validate(epoch)
            
            # 学习率调度
            self.scheduler.step(val_loss)
            
            # 记录损失
            self.train_losses.append(train_loss)
            self.val_losses.append(val_loss)
            
            epoch_time = time.time() - start_time
            
            # 日志
            log_message = (f"Epoch {epoch}/{self.config.epochs}, "
                         f"Train Loss: {train_loss:.6f}, Val Loss: {val_loss:.6f}, "
                         f"Time: {epoch_time:.2f}s, LR: {self.optimizer.param_groups[0]['lr']:.2e}")
            self.log(log_message)
            
            # 保存最佳模型
            if val_loss < self.best_val_loss:
                self.best_val_loss = val_loss
                self.save_model(self.config.model_save_path)
                self.log(f"保存最佳模型，验证损失: {val_loss:.6f}")
            
            # 每10个epoch保存一次检查点
            if epoch % 10 == 0:
                checkpoint_path = f"model/checkpoint_epoch_{epoch}.pth"
                self.save_model(checkpoint_path)
                self.log(f"保存检查点: {checkpoint_path}")

if __name__ == "__main__":
    config = Config()
    trainer = Trainer(config)
    trainer.train()

import torch
import torchaudio
import os
from pathlib import Path
import argparse
import numpy as np
import sys
from tqdm import tqdm

# 添加src目录到路径
sys.path.append('../src')

from config import Config
from model import GuitarSeparator

class GuitarSeparatorInference:
    def __init__(self, model_path, device="cuda" if torch.cuda.is_available() else "cpu"):
        self.config = Config()
        self.device = device
        
        print("正在加载模型...")
        # 加载模型
        self.model = GuitarSeparator(self.config).to(self.device)
        
        checkpoint = torch.load(model_path, map_location=device)
        self.model.load_state_dict(checkpoint['model_state_dict'])
        self.model.eval()
        
        print(f"模型加载成功: {model_path}")
    
    def preprocess_audio(self, audio_path):
        """预处理音频文件"""
        print(f"正在预处理音频: {audio_path}")
        audio, sr = torchaudio.load(audio_path)
        
        # 重采样到目标采样率
        if sr != self.config.sample_rate:
            print(f"重采样: {sr}Hz -> {self.config.sample_rate}Hz")
            resampler = torchaudio.transforms.Resample(sr, self.config.sample_rate)
            audio = resampler(audio)
        
        # 转换为立体声
        if audio.shape[0] == 1:
            audio = audio.repeat(2, 1)
        elif audio.shape[0] > 2:
            audio = audio[:2, :]
        
        print(f"音频形状: {audio.shape}")
        return audio
    
    def separate_guitar(self, audio_path, output_path=None):
        """分离吉他音轨"""
        # 预处理音频
        audio = self.preprocess_audio(audio_path)
        original_length = audio.shape[1]
        
        # 如果音频太长，分段处理
        segment_length = int(self.config.sample_rate * 10)  # 10秒一段
        overlap = int(self.config.sample_rate * 2)  # 2秒重叠
        
        if audio.shape[1] > segment_length:
            segments = []
            start = 0
            
            # 计算段数
            num_segments = (audio.shape[1] - overlap) // (segment_length - overlap)
            if (audio.shape[1] - overlap) % (segment_length - overlap) > 0:
                num_segments += 1
            
            print(f"音频过长，分段处理: {num_segments} 段")
            
            # 分割音频
            for i in range(num_segments):
                start = i * (segment_length - overlap)
                end = min(start + segment_length, audio.shape[1])
                segment = audio[:, start:end]
                
                # 如果段太短，填充
                if segment.shape[1] < segment_length:
                    pad_length = segment_length - segment.shape[1]
                    segment = torch.nn.functional.pad(segment, (0, pad_length))
                
                segments.append(segment)
            
            # 处理每个段
            guitar_segments = []
            for i, segment in enumerate(tqdm(segments, desc="处理音频段")):
                with torch.no_grad():
                    segment_guitar = self.model(segment.unsqueeze(0).to(self.device))
                    segment_guitar = segment_guitar.squeeze(0).cpu()
                
                # 移除填充（如果有）
                actual_end = min(segment_length, original_length - start)
                segment_guitar = segment_guitar[:, :actual_end]
                
                guitar_segments.append((start, segment_guitar))
            
            # 合并段（使用交叉淡化处理重叠区域）
            guitar_audio = self._merge_segments(guitar_segments, overlap, original_length)
        else:
            # 直接处理短音频
            print("直接处理短音频")
            with torch.no_grad():
                guitar_audio = self.model(audio.unsqueeze(0).to(self.device))
                guitar_audio = guitar_audio.squeeze(0).cpu()
        
        # 保存结果
        if output_path is None:
            input_dir = os.path.dirname(audio_path)
            input_name = os.path.splitext(os.path.basename(audio_path))[0]
            output_path = os.path.join(input_dir, f"{input_name}_guitar.wav")
        
        print(f"保存吉他音轨: {output_path}")
        torchaudio.save(output_path, guitar_audio, self.config.sample_rate)
        
        return guitar_audio
    
    def _merge_segments(self, segments, overlap, total_length):
        """合并音频段，使用交叉淡化处理重叠"""
        print("合并音频段...")
        merged = torch.zeros((2, total_length))
        
        for i, (start, segment) in enumerate(tqdm(segments, desc="合并段")):
            seg_length = segment.shape[1]
            end = start + seg_length
            
            if i == 0:
                # 第一段
                merged[:, start:end] = segment
            else:
                # 重叠区域交叉淡化
                overlap_start = start
                overlap_end = start + overlap
                
                if overlap_end > total_length:
                    overlap_end = total_length
                
                overlap_length = overlap_end - overlap_start
                
                if overlap_length > 0:
                    fade_out = torch.linspace(1, 0, overlap_length)
                    fade_in = torch.linspace(0, 1, overlap_length)
                    
                    # 应用交叉淡化
                    merged[:, overlap_start:overlap_end] = (
                        merged[:, overlap_start:overlap_end] * fade_out + 
                        segment[:, :overlap_length] * fade_in
                    )
                
                # 添加剩余部分
                non_overlap_start = overlap_end
                non_overlap_end = end
                
                if non_overlap_start < non_overlap_end:
                    merged[:, non_overlap_start:non_overlap_end] = segment[:, overlap_length:overlap_length + (non_overlap_end - non_overlap_start)]
        
        return merged

def process_directory(input_dir, output_dir, model_path):
    """处理目录中的所有音频文件"""
    separator = GuitarSeparatorInference(model_path)
    
    # 支持的音频格式
    audio_extensions = {'.mp3', '.wav', '.flac', '.m4a', '.aac'}
    
    # 获取所有音频文件
    audio_files = []
    for ext in audio_extensions:
        audio_files.extend(Path(input_dir).glob(f"*{ext}"))
        audio_files.extend(Path(input_dir).glob(f"*{ext.upper()}"))
    
    print(f"找到 {len(audio_files)} 个音频文件")
    
    for audio_file in tqdm(audio_files, desc="处理文件"):
        try:
            output_file = Path(output_dir) / f"{audio_file.stem}_guitar.wav"
            separator.separate_guitar(str(audio_file), str(output_file))
        except Exception as e:
            print(f"处理文件 {audio_file} 时出错: {e}")
